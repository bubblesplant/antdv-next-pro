Components/ProDescriptions

ProDescriptions ​

Display a single record with table-compatible field schemas, formatting, requests and field editing. Layout props follow the native antdv-next Descriptions component.

vue
<script setup lang="ts">
import { ref } from 'vue'
import { Button, Space } from 'antdv-next'
import {
  ProDescriptions,
  type ProDescriptionsColumn,
  type ProDescriptionsInstance,
  type ProKey,
} from '@bubblesjs/antdv-next-pro'

interface ProjectDetails extends Record<string, unknown> {
  name: string
  owner: string
  status: string
  budget: number
  createdAt: string
  notes: string
}

const action = ref<ProDescriptionsInstance<ProjectDetails>>()
const server = ref<ProjectDetails>({
  name: '组件升级计划',
  owner: '林晓',
  status: 'active',
  budget: 16800,
  createdAt: '2026-10-21',
  notes: '先完成示例，再安排团队验收。',
})
const requests = ref(0)
const feedback = ref('点击字段旁的编辑图标；清空项目名称后保存可查看校验。')
const columns: ProDescriptionsColumn<ProjectDetails>[] = [
  {
    title: '项目名称',
    dataIndex: 'name',
    formItemProps: { rules: [{ required: true, message: '请输入项目名称' }] },
  },
  { title: '负责人', dataIndex: 'owner' },
  {
    title: '状态',
    dataIndex: 'status',
    valueType: 'select',
    valueEnum: {
      active: { text: '进行中', status: 'Processing' },
      done: { text: '已完成', status: 'Success' },
      pending: { text: '待开始', status: 'Default' },
    },
  },
  { title: '预算', dataIndex: 'budget', valueType: 'money', editable: false },
  { title: '创建日期', dataIndex: 'createdAt', valueType: 'date', editable: false },
  { title: '备注', dataIndex: 'notes', valueType: 'textarea' },
]

async function request() {
  requests.value++
  return { data: { ...server.value }, success: true }
}
async function save(_key: ProKey, record: ProjectDetails) {
  server.value = { ...record }
  feedback.value = `已保存：${record.name} · ${record.owner}`
  return true
}
</script>

<template>
  <div class="descriptions-demo" data-demo="pro-descriptions">
    <ProDescriptions
      ref="action"
      title="项目详情"
      tooltip="通过 request 获取详情，逐字段编辑并保存。"
      :columns="columns"
      :request="request"
      :editable="{ type: 'single', onSave: save }"
      :column="{ xs: 1, sm: 2 }"
      bordered
      @editable-error="feedback = '请检查字段内容后重试。'"
    >
      <template #extra>
        <Space>
          <Button data-demo-descriptions-edit @click="action?.startEditable('name')"
            >编辑名称</Button
          >
          <Button data-demo-descriptions-reload @click="action?.reload()">重新加载</Button>
        </Space>
      </template>
    </ProDescriptions>
    <p data-demo-descriptions-state aria-live="polite">已加载 {{ requests }} 次。{{ feedback }}</p>
    <pre data-demo-descriptions-data>{{ JSON.stringify(server, null, 2) }}</pre>
  </div>
</template>

<style scoped>
.descriptions-demo {
  display: grid;
  gap: 16px;
}
p {
  margin: 0;
}
pre {
  margin: 0;
  overflow: auto;
  padding: 16px;
  border: 1px solid #d9d9d9;
  border-radius: 8px;
}
</style>

Minimal usage ​

vue
<script setup lang="ts">
import { ref } from 'vue'
import { ProDescriptions, type ProDescriptionsColumn } from '@bubblesjs/antdv-next-pro'

type Profile = { name: string; status: string }
const profile = ref<Profile>({ name: 'Alex', status: 'active' })
const columns: ProDescriptionsColumn<Profile>[] = [
  { title: 'Name', dataIndex: 'name', formItemProps: { rules: [{ required: true }] } },
  {
    title: 'Status',
    dataIndex: 'status',
    valueType: 'select',
    valueEnum: { active: 'Active', inactive: 'Inactive' },
  },
]
</script>

<template>
  <ProDescriptions
    title="Profile"
    :columns="columns"
    :data-source="profile"
    :editable="{ type: 'single' }"
    @data-source-change="profile = $event"
  />
</template>

The type inherits ProColumns. Table-only entries (width/minWidth/fixed/align/sorter/filters/hideInTable/hideInSearch/hideInForm/search/colSize), nested columns/children, initialValue, convertValue and transform are not processed by this description renderer. Configure the native form or a custom renderer when needed.

The edit context contains record/index/column/editable/isEditable/recordKey/type="descriptions"/dependencies/defaultRender()/update(value). form is the actual native FormInstance.

Editable configuration ​

PropertyTypeDefaultDescription / source
type'single' | 'multiple'singleAllows one or multiple fields to be edited.
editableKeysProKey[]—Controlled editing keys; update them in onChange.
onChange(keys: ProKey[], rows: T[]) => void—Editing keys changed; rows contains the single full draft.
onValuesChange(record: T, rows: T[]) => void—Draft update; rows=[record].
onSave(key, next: T, origin: T) => void | boolean | Promise<void | boolean>—Validates and merges only this field; false or errors retain editing.
onCancel(key, draft: T, origin: T) => void | Promise<void>—Cancels a field and restores its original value.
formPropsFormProps—Native Form props, overriding top-level formProps; internal model wins.
formRefRef<FormInstance | undefined>—Receives the actual native form, like top-level formRef.

The shared EditableConfig type also contains table-only onDelete/actionRender/form and action-text/alert options. ProDescriptions does not consume them; its save/cancel buttons use Pro locale text. Save validates only the current field. onSave=false or validation errors keep editing active; controlled keys still require parent updates.

Native item member ​

ProDescriptions.Item and ProDescriptionsItem alias native Descriptions.Item. Its key/label/children/span and native slots follow that API; it adds no Pro-specific events or instance methods. Default-slot native items do not become editable schema fields.
