Components/QueryFilter

QueryFilter ​

QueryFilter is a standalone search form with responsive columns, search and reset actions, and expandable fields. ProTable uses the same component for its default search area. Compose ProForm fields or pass columns to generate fields from a schema.

Import ​

ts
import { QueryFilter, ProFormText, ProFormSelect } from '@bubblesjs/antdv-next-pro'
import '@bubblesjs/antdv-next-pro/style.css'

You can also register <QueryFilter> globally with app.use(AntdvNextPro). See Getting started.

Fields and expansion ​

Place ProFormText, ProFormSelect, and other fields in the default slot. Bind values with v-model and handle validated searches through :on-finish="search".

View full source
vue
<script setup lang="ts">
import { ref } from 'vue'
import { ProFormSelect, ProFormText, QueryFilter } from '@bubblesjs/antdv-next-pro'

const initialValues = { status: 'active' }
const model = ref<Record<string, unknown>>({ ...initialValues })
const collapsed = ref(true)
const submitted = ref('')
const count = ref(0)

function search(values: Record<string, unknown>) {
  submitted.value = JSON.stringify(values, null, 2)
  count.value++
  return true
}
</script>

<template>
  <div>
    <QueryFilter
      v-model="model"
      v-model:collapsed="collapsed"
      :initial-values="initialValues"
      :span="8"
      :label-width="80"
      :on-finish="search"
      @reset="submitted = ''"
    >
      <ProFormText name="keyword" label="项目名称" :field-props="{ placeholder: '请输入关键词' }" />
      <ProFormSelect
        name="status"
        label="项目状态"
        :options="[
          { label: '进行中', value: 'active' },
          { label: '已完成', value: 'done' },
        ]"
      />
      <ProFormText name="owner" label="负责人" :field-props="{ placeholder: '展开后填写负责人' }" />
      <ProFormSelect
        name="priority"
        label="优先级"
        :options="[
          { label: '普通', value: 'normal' },
          { label: '紧急', value: 'urgent' },
        ]"
      />
    </QueryFilter>
    <div class="filter-result" aria-live="polite">
      <p>查询区{{ collapsed ? '已收起' : '已展开' }} · 已查询 {{ count }} 次</p>
      <span>{{ submitted ? '最近一次查询参数' : '当前表单值' }}</span>
      <pre>{{ submitted || JSON.stringify(model, null, 2) }}</pre>
    </div>
  </div>
</template>

<style scoped>
.filter-result {
  margin-top: 20px;
  padding: 16px;
  border: 1px solid #e8ecf1;
  border-radius: 6px;
  background: #fafbfc;
  color: #64748b;
  font-size: 12px;
}
.filter-result p {
  margin: 0 0 12px;
}
.filter-result pre {
  margin: 8px 0 0;

}
</style>

defaultCollapsed defaults to true. Use v-model:collapsed for controlled expansion. When fields exceed the first row's capacity, the collapse action appears with a down arrow to expand and an up arrow to collapse. Hidden fields retain their values.

span uses a 24-column grid and accepts a number or breakpoint map, for example { xs: 24, sm: 12, md: 8 }. Without an explicit span, column widths follow the container width and defaultColsNumber.

Schema fields and instance methods ​

Pass existing schema columns directly. A component ref can set values, submit, and reset independently of a table.

View full source
vue
<script setup lang="ts">
import { ref } from 'vue'
import { Button, Space } from 'antdv-next'
import {
  QueryFilter,
  type SchemaFormColumn,
  type SchemaFormInstance,
} from '@bubblesjs/antdv-next-pro'

type Filters = Record<string, unknown> & { keyword?: string; owner?: string; status?: string }
const form = ref<SchemaFormInstance<Filters>>()
const model = ref<Partial<Filters>>({})
const submitted = ref('')
const columns: SchemaFormColumn<Filters>[] = [
  { title: '项目名称', dataIndex: 'keyword', fieldProps: { placeholder: '请输入项目名称' } },
  {
    title: '项目状态',
    dataIndex: 'status',
    valueType: 'select',
    valueEnum: { active: '进行中', done: '已完成' },
  },
  { title: '负责人', dataIndex: 'owner', fieldProps: { placeholder: '请输入负责人' } },
]

function search(values: Partial<Filters>) {
  submitted.value = JSON.stringify(values, null, 2)
  return true
}
</script>

<template>
  <div>
    <Space style="margin-bottom: 16px" wrap>
      <Button @click="form?.setFieldsValue({ keyword: '组件升级', status: 'active' })">
        填入示例条件
      </Button>
      <Button @click="form?.submit()">通过 ref 查询</Button>
      <Button @click="form?.reset()">通过 ref 重置</Button>
    </Space>
    <QueryFilter
      ref="form"
      v-model="model"
      :columns="columns"
      :default-collapsed="false"
      :on-finish="search"
      @reset="submitted = ''"
    />
    <pre class="filter-result" aria-live="polite">{{
      submitted || JSON.stringify(model, null, 2)
    }}</pre>
  </div>
</template>

<style scoped>
.filter-result {
  margin-top: 20px;
  overflow: auto;
  padding: 16px;
  border: 1px solid #e8ecf1;
  border-radius: 6px;
  background: #fafbfc;
  color: #334155;
  font-size: 12px;
}
</style>

See SchemaForm for column configuration. These examples demonstrate composition and schema fields separately.

API ​

The component also supports the shared ProForm options, including initialValues, request, params, validation, value formatting, and URL synchronization.

PropertyTypeDefault / description
modelValue / v-modelPartial<T>Current search values
columnsSchemaFormColumn<T>[]Optional schema fields
onFinish(values) => boolean | void | Promise<boolean | void>Validated search callback; bind with :on-finish
collapsed / v-model:collapsedbooleanControlled collapsed state
defaultCollapsedbooleantrue
defaultColsNumbernumber3; default column count in wider containers
spannumber | Record<string, number>Grid span with xs/sm/md/lg/xl breakpoints
labelWidthnumber | 'auto'Label width; numbers use px
splitbooleanfalse; separator styling
searchText / resetTextVNodeChildLocalized search / reset labels
optionRenderfalse | (props, form, doms) => VNodeChildCustomize search and reset actions; false hides them
collapseRenderfalse | (collapsed) => VNodeChildCustomize collapse content; false hides the collapse action
submitterfalse | SubmitterPropsConfigure search and reset actions and rendering

optionRender=false keeps the collapse action. Set collapseRender=false to hide it. A custom collapseRender replaces the default text and arrow.

EventDescription
update:modelValue / changeSearch values changed
update:collapsed / collapseReceives the new collapsed state
submitObserve submission; use onFinish to await business results
resetRestore the initialization snapshot
error / requestErrorSubmission error / initialization request error

The default slot composes fields. Type the component ref as SchemaFormInstance<T> to access getFieldsValue(), setFieldsValue(), validate(), submit(), reset(), and other ProForm instance methods.
