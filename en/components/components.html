Components/Toolbar and helper components

Toolbar and helper components ​

Composable toolbar, action menu, search form, filter popup, field label and error components. All are exported from @bubblesjs/antdv-next-pro.

vue
<script setup lang="ts">
import { computed, defineComponent, h, ref } from 'vue'
import { Button, Form, Input, Space } from 'antdv-next'
import {
  DropdownFooter,
  ErrorBoundary,
  FieldLabel,
  FilterDropdown,
  InlineErrorFormItem,
  LabelIconTip,
  LightWrapper,
  ListToolBar,
  ProForm,
  ProFormText,
  Search,
  TableDropdown,
  type ProColumns,
} from '@bubblesjs/antdv-next-pro'

const DropdownButton = TableDropdown.Button
const feedback = ref('操作工具栏、筛选标签与错误恢复，结果会显示在这里。')
const dropdownOpen = ref(false)
const filterDraft = ref('进行中')
const filterValue = ref<string>()
const name = ref('')
const inlineModel = computed(() => ({ name: name.value }))
const showError = ref(false)
const error = computed(() =>
  showError.value && name.value.trim().length < 2 ? '请至少输入两个字符' : undefined,
)
const crash = ref(false)
const CrashContent = defineComponent({
  props: { fail: Boolean },
  setup(props) {
    return () => {
      if (props.fail) throw new Error('这是用于演示恢复流程的错误。')
      return h(
        'p',
        { 'data-demo-boundary-content': '' },
        '内容正常，点击“触发错误”查看自定义降级界面。',
      )
    }
  },
})
const searchColumns: ProColumns[] = [
  { title: '项目名称', dataIndex: 'name' },
  {
    title: '状态',
    dataIndex: 'status',
    valueType: 'select',
    valueEnum: { active: '进行中', done: '已完成' },
  },
]
const filterFooter = {
  onConfirm: () => {
    filterValue.value = filterDraft.value
    feedback.value = `已应用筛选：${filterDraft.value}`
    dropdownOpen.value = false
  },
  onClear: () => {
    filterDraft.value = ''
    filterValue.value = undefined
    feedback.value = '已清除筛选'
  },
}
async function submitLight(values: Record<string, unknown>) {
  feedback.value = `轻量筛选：${JSON.stringify(values)}`
  return true
}
function validateName() {
  showError.value = true
  feedback.value = name.value.trim().length >= 2 ? '名称校验通过' : '名称校验未通过'
}
function recover(reset: () => void) {
  crash.value = false
  reset()
}
</script>

<template>
  <div class="shared-demo" data-demo="shared-components">
    <section>
      <ListToolBar
        title="独立工具栏"
        sub-title="下拉操作与搜索"
        tooltip="此处展示公开的辅助组件。"
        :search="{
          placeholder: '输入工具栏关键词',
          onSearch: (value: string) => {
            feedback = `工具栏搜索：${value}`
          },
        }"
      >
        <template #actions>
          <TableDropdown
            text="更多操作"
            :menus="[
              { key: 'export', name: '导出列表' },
              { key: 'archive', name: '归档项目' },
            ]"
            :on-select="
              (key: string) => {
                feedback = `选中菜单：${key}`
              }
            "
          />
          <DropdownButton
            text="按钮菜单"
            :menus="[{ key: 'copy', name: '复制链接' }]"
            :on-select="
              (key: string) => {
                feedback = `按钮菜单：${key}`
              }
            "
            data-demo-dropdown-button
          />
        </template>
      </ListToolBar>
      <Search
        :columns="searchColumns"
        :search="{ defaultCollapsed: false, searchText: '应用查询' }"
        @submit="feedback = `查询条件：${JSON.stringify($event)}`"
        @reset="feedback = '查询已重置'"
      />
    </section>

    <section>
      <h3>
        <LabelIconTip
          label="筛选与提示"
          sub-title="点击标签打开筛选"
          tooltip="确认应用修改，取消保留上次结果。"
        />
      </h3>
      <Space wrap>
        <FilterDropdown
          v-model:open="dropdownOpen"
          label="状态筛选"
          :footer="filterFooter"
          placement="bottomLeft"
        >
          <template #label><Button data-demo-filter-open>状态筛选</Button></template>
          <Input v-model:value="filterDraft" aria-label="筛选状态" data-demo-filter-input />
        </FilterDropdown>
        <FieldLabel
          label="已选状态"
          :value="filterValue"
          :on-clear="
            () => {
              filterValue = undefined
              feedback = '已清除标签'
            }
          "
        />
      </Space>
      <ProForm
        :submitter="false"
        :initial-values="{ keyword: '组件' }"
        :on-finish="submitLight"
        class="light-form"
      >
        <LightWrapper name="keyword" label="关键词">
          <ProFormText
            name="keyword"
            placeholder="输入关键词"
            :rules="[{ max: 12, message: '最多输入 12 个字符' }]"
          />
        </LightWrapper>
      </ProForm>
      <div class="footer-preview">
        <span>独立 DropdownFooter</span>
        <DropdownFooter
          clear-text="清除结果"
          confirm-text="确认结果"
          :on-clear="
            () => {
              feedback = '独立操作区：已清除'
            }
          "
          :on-confirm="
            () => {
              feedback = '独立操作区：已确认'
            }
          "
        />
      </div>
    </section>

    <section>
      <h3>行内校验</h3>
      <Form :model="inlineModel" layout="vertical">
        <InlineErrorFormItem label="显示名称" name="name" :error="error">
          <Input v-model:value="name" placeholder="至少两个字符" data-demo-inline-input />
        </InlineErrorFormItem>
        <Button data-demo-inline-validate @click="validateName">校验名称</Button>
      </Form>
    </section>

    <section>
      <h3>错误边界</h3>
      <Button data-demo-boundary-crash :disabled="crash" @click="crash = true">触发错误</Button>
      <ErrorBoundary
        :on-error="
          () => {
            feedback = '错误已被边界捕获，可点击恢复'
          }
        "
      >
        <CrashContent :fail="crash" />
        <template #fallback="{ error: boundaryError, reset }">
          <div role="alert" class="fallback">
            <p>{{ boundaryError.message }}</p>
            <Button data-demo-boundary-reset @click="recover(reset)">恢复内容</Button>
          </div>
        </template>
      </ErrorBoundary>
    </section>
    <p data-demo-shared-state aria-live="polite">{{ feedback }}</p>
  </div>
</template>

Minimal usage ​

vue
<script setup lang="ts">
import { ref } from 'vue'
import { Button, Input } from 'antdv-next'
import { LightWrapper, ListToolBar, TableDropdown } from '@bubblesjs/antdv-next-pro'

const open = ref(false)
const keyword = ref('')
const selected = ref('')
const menus = [
  { key: 'export', name: 'Export' },
  { key: 'archive', name: 'Archive' },
]
</script>

<template>
  <ListToolBar title="Projects">
    <template #actions>
      <TableDropdown :menus="menus" @select="selected = $event" />
      <Button type="primary">Create</Button>
    </template>
  </ListToolBar>
  <LightWrapper
    v-model:open="open"
    label="Keyword"
    :value="keyword"
    allow-clear
    @clear="keyword = ''"
  >
    <Input v-model:value="keyword" />
  </LightWrapper>
  <p>Selected menu: {{ selected }}</p>
</template>

ListToolBar ​

PropBehavior
title / subTitle / tooltipHeading, subtitle and tooltip
actions / settingsNodes or actions(), and an array of settings nodes
searchBoolean or native InputSearch props; supports asynchronous onSearch(value)
menu{ type?, items?, activeKey?, defaultActiveKey?, onChange? }; inline, tab or dropdown
tabs{ items?, activeKey?, defaultActiveKey?, onChange? } on an extra line
filter / multipleLineFilter content and multiline styling

Returning false from search.onSearch suppresses the component search(value) event. Menu/tabs onChange(key) notify selection changes; accept controlled activeKey updates in the parent.

Slots: title, actions, settings, filter. Semantic parts: root, container, left, right, title, subTitle, extraLine. No separate instance methods.

TableDropdown / TableDropdown.Button ​

PropBehavior
menus / options{ key: string, name?, label?, disabled?, danger? }[]; menus takes precedence
textTrigger text
disabledDisable the menu
onSelect(key) / @selectString key of an enabled menu item

Label takes precedence over name. The default slot replaces the regular TableDropdown trigger; otherwise it uses an ellipsis icon. TableDropdown.Button is a real component member, also exported as TableDropdownButton. It uses a native Button trigger and forwards additional Button attrs. Semantic part: root. No separate instance methods.

Search ​

A standalone schema search form. Search filterType light uses LightFilter; otherwise QueryFilter is used.

Prop/eventBehavior
columnsProColumns; excludes hideInSearch=true and search=false entries
searchProTable search configuration: span, labelWidth, collapsed, defaultCollapsed, searchText, resetText, optionRender and collapseRender
formNative Form props plus initialValues and submitter; classes/styles customize the filter form
formRefRef<SchemaFormInstance<T> | undefined>
submit(values) / reset() / collapse(collapsed)Submit, reset and collapse notifications

Loading controls the default submit button. Search embedded in ProTable receives table request loading automatically. Explicit form.submitter.submitButtonProps.loading takes precedence; submitter=false and submitButtonProps=false still hide the corresponding actions.

The component ref forwards SchemaFormInstance. Search collects and validates values; the caller performs requests. There are no extra business slots or a separate semantic part map.

FilterDropdown ​

Prop/eventBehavior
label / disabled / placementTrigger, disabled state and popup position
open / v-model:openInternal state when omitted; otherwise accept update:open in the parent
popoverPropsNative Popover props; onOpenChange is merged with component behavior
footerDropdownFooter configuration; passing it enables the action area
footerRenderFalse hides the footer; a function replaces its content
paddingContent padding
openChange(open)Requested open-state changes

The default slot contains the panel, and label replaces the trigger. Semantic parts are root for the trigger and content for the panel. Customize the popup container with popoverProps.styles.container. Footer callbacks own clearing and closing behavior; no data changes happen automatically. No separate instance methods.

LightWrapper ​

Combines FieldLabel and FilterDropdown. Props: label, value, open, disabled, allowClear, footerRender, placement. The default slot is panel content. Semantic parts: root, content.

Events: update:open, clear, confirm. Confirm emits and requests closing; clear only notifies the parent, which updates value. Open is internal when omitted and controlled with v-model:open. No separate instance methods.

DropdownFooter ​

Props: disabled, confirmText, clearText, onConfirm(event?), onClear(event?), footerRender. No business slots or instance methods.

FooterRender false renders nothing. A function receives (confirm?, clear?) and returns VNodeChild. Callbacks only notify; they do not wait for asynchronous operations, clear data or close a parent. Semantic parts: root, clear, confirm. Custom footerRender replaces the default buttons, so their parts no longer apply.

FieldLabel / LabelIconTip ​

ComponentProps and behaviorSlotsSemantic parts
FieldLabellabel, value, disabled, expanded, ellipsis, onClear; non-null values are displayed; expanded supplies accessibility stateNoneroot, label, value
LabelIconTiplabel, tooltip (string or TooltipProps), subTitle, ellipsisDefault replaces labelroot, label, subTitle, tooltip

FieldLabel uses a focusable button. A value with onClear shows a clear control that stops propagation; otherwise an arrow is shown. Panel state belongs to the caller. Neither exposes separate instance methods.

InlineErrorFormItem ​

Extends native FormItem props with error and errorType default or popover. Default mode shows inline errors. Popover mode uses the truthiness of error to open the popup and hides inline help.

The default slot contains the control; error customizes inline or popup error content. An error slot alone does not open a popup when error is empty. Semantic parts: root, field (native FormItem). Native Form/FormItem still owns validation rules. No separate instance methods.

ErrorBoundary ​

Catches rendering and lifecycle errors in descendant Vue components and shows fallback content.

APIBehavior
fallbackA node or (error, reset) => VNodeChild
resetKeysReset when array elements change
onError(error, info)Error notification
Default slotNormal content
fallback slot{ error, reset }, takes precedence over the prop
ref.reset()Clear the error and render descendants again

The default fallback uses native Result and a retry button. The root semantic part exists only in the fallback state; normal rendering directly returns the default slot. Handle errors in the boundary itself, detached asynchronous Promises and network operations at their source.

Shared styling and configuration ​

Except Search, which uses its form configuration, these components provide prefixCls, classes and styles. Classes/styles accept a map or ({ props }) => map. Legacy classNames remains compatible; classes wins for matching parts. Only documented parts are consumed.

ComponentDefault public prefix
ListToolBarant-pro-table-list-toolbar
TableDropdown / TableDropdown.Buttonant-pro-table-dropdown
LightWrapperant-pro-field-light-wrapper
FilterDropdownant-pro-core-field-dropdown
DropdownFooterant-pro-core-field-dropdown-footer
FieldLabelant-pro-core-field-label
LabelIconTipant-pro-core-label-tip
InlineErrorFormItemant-pro-form-inline-error-form-item
ErrorBoundaryant-pro-error-boundary

Native ConfigProvider prefixCls="demo" replaces ant-pro with demo-pro in these prefixes. The component prefixCls can override it.

Locale, direction, theme algorithms and native tokens follow ConfigProvider. Configure Pro tokens through ProConfigProvider.token. Native Result tokens use theme.components.Result; helpers do not register independent theme.components.ProX entries.
